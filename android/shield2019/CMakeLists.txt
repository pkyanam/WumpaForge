cmake_minimum_required(VERSION 3.20)
project(wumpaforge_shield LANGUAGES C CXX)
include(../probe/target_contract.cmake)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(WUMPA_ROOT ../.. ABSOLUTE)
set(WUMPA_WORK "${WUMPA_ROOT}/build/shield2019")
set(RUNTIME "${WUMPA_WORK}/runtime/src")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_subdirectory("${WUMPA_WORK}/deps/SDL-5d249570393f7a37e037abf22cd6012a4cc56a71" sdl2)
file(GLOB KERNEL "${RUNTIME}/kernel/*.c")
set(RUNTIME_SOURCES ${RUNTIME}/platform/win32_compat.c ${KERNEL}
    ${RUNTIME}/d3d/d3d8_gl.c ${RUNTIME}/audio/dsound_device.c ${RUNTIME}/audio/xbox_adpcm.c
    ${RUNTIME}/input/xinput_device.c ${RUNTIME}/usb/ohci.c ${RUNTIME}/usb/usb_gamepad.c)
foreach(FILE apu_core apu_vp apu_dsp apu_mmio_hook apu_xaudio2)
    list(APPEND RUNTIME_SOURCES ${RUNTIME}/apu/${FILE}.c)
endforeach()
foreach(FILE nv2a_core nv2a_pgraph_d3d11 nv2a_pb_replay nv2a_mmio_hook nv2a_pb_test)
    list(APPEND RUNTIME_SOURCES ${RUNTIME}/nv2a/${FILE}.c)
endforeach()
foreach(FILE video_player video_pump fb_present)
    list(APPEND RUNTIME_SOURCES ${RUNTIME}/video/${FILE}.c)
endforeach()
add_library(shield_runtime STATIC ${RUNTIME_SOURCES} ${WUMPA_WORK}/gl/gl_loader.c)
target_compile_definitions(shield_runtime PUBLIC _GNU_SOURCE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(shield_runtime PUBLIC ${RUNTIME} ${CMAKE_CURRENT_SOURCE_DIR} ${WUMPA_WORK}/gl)
foreach(DIR platform kernel d3d audio input usb apu nv2a video)
    target_include_directories(shield_runtime PUBLIC ${RUNTIME}/${DIR})
endforeach()
target_compile_options(shield_runtime PRIVATE -O2 -fno-strict-aliasing -fwrapv -Werror=implicit-function-declaration)
target_link_libraries(shield_runtime PUBLIC SDL2::SDL2 android log EGL dl m)
file(GLOB GENERATED CONFIGURE_DEPENDS "${WUMPA_ROOT}/local/generated/*.c")
if(NOT GENERATED)
    message(FATAL_ERROR "Generate the original AOT game sources with the host pipeline first.")
endif()
file(GLOB TITLE "${WUMPA_WORK}/title/*.c")
add_library(main SHARED entry.c host.c ${TITLE} ${GENERATED})
target_include_directories(main PRIVATE "${WUMPA_ROOT}/local/generated")
target_link_libraries(main PRIVATE shield_runtime)
target_link_options(main PRIVATE "-Wl,--no-undefined")
target_compile_options(main PRIVATE -fno-strict-aliasing -fwrapv -Wno-unused-label -Wno-unused-variable)
set_source_files_properties(${GENERATED} PROPERTIES COMPILE_OPTIONS "-O1;-gline-tables-only")
set_source_files_properties(${WUMPA_WORK}/title/main.c PROPERTIES COMPILE_DEFINITIONS "main=wrath_game_main")

add_library(shield_probe SHARED diagnostics.c ../probe/memory_probe.c ../probe/android_backing.c)
target_compile_definitions(shield_probe PRIVATE _GNU_SOURCE)
set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/../probe/memory_probe.c PROPERTIES COMPILE_DEFINITIONS "main=wumpa_memory_probe")
target_include_directories(shield_probe PRIVATE ${WUMPA_WORK}/gl)
target_link_libraries(shield_probe PRIVATE EGL log android)
target_link_options(shield_probe PRIVATE "-Wl,--no-undefined")

add_library(graphics_check SHARED graphics_check_entry.c host.c
    ${WUMPA_WORK}/checks/graphics_check.c
    ${WUMPA_WORK}/title/nv2a_vertex.c ${WUMPA_WORK}/title/nv2a_vertex_input.c ${WUMPA_WORK}/title/nv2a_pixel.c)
target_compile_definitions(graphics_check PRIVATE WRATH_GRAPHICS_SMOKE_TEST)
target_compile_options(graphics_check PRIVATE -UNDEBUG -O1)
target_include_directories(graphics_check PRIVATE ${WUMPA_WORK}/title)
target_link_libraries(graphics_check PRIVATE shield_runtime)
target_link_options(graphics_check PRIVATE "-Wl,--no-undefined")

add_library(controller_check SHARED controller_check.c host.c)
target_compile_options(controller_check PRIVATE -UNDEBUG)
target_link_libraries(controller_check PRIVATE shield_runtime)
target_link_options(controller_check PRIVATE "-Wl,--no-undefined")
